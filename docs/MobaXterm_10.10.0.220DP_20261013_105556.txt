#!/bin/sh
#
# rc.local
#
# This script is executed at the end of each multiuser runlevel.
# Make sure that the script will "exit 0" on success or any other
# value on error.
#
# In order to enable or disable this script just change the execution
# bits.
#
# By default this script does nothing.

/usr/local/bin/gen-friendlyelec-release
. /etc/friendlyelec-release
if [ ! -f /etc/firstuse ]; then
    /bin/echo ${BOARD} > /etc/hostname
    /bin/sed -i "s/\(127.0.1.1\s*\).*/\1${BOARD}/g" /etc/hosts
    /bin/hostname ${BOARD}
    /bin/echo "0" > /etc/firstuse
fi

#/bin/echo "nameserver 8.8.8.8" /etc/resolv.conf

#################################################################
# 자동 파티션 확장 스크립트 (섹터 수 기반 판단)
#################################################################

# 로그 파일 초기화
echo "$(date): Auto-expand script started" > /tmp/auto-expand.log

# 첫 부팅 시 파티션 확장
if [ ! -f /.filesystem_expanded ]; then
    echo "$(date): First boot detected - checking partition size" >> /tmp/auto-expand.log

    # 현재 루트 파티션 크기 확인 (KB 단위)
    ROOT_SIZE=$(df / | tail -1 | awk '{print $2}')
    TOTAL_DISK_SIZE=$(fdisk -l /dev/mmcblk0 | grep "Disk /dev/mmcblk0" | awk '{print $5}')
    TOTAL_DISK_GB=$((TOTAL_DISK_SIZE / 1024 / 1024 / 1024))

    echo "$(date): Current root partition size: ${ROOT_SIZE}KB" >> /tmp/auto-expand.log
    echo "$(date): Total disk size: ${TOTAL_DISK_GB}GB" >> /tmp/auto-expand.log

    # 디스크 전체 크기의 80% 미만이면 확장 필요
    EXPECTED_SIZE=$((TOTAL_DISK_SIZE * 80 / 100 / 1024))  # 80%를 KB로 계산
    echo "$(date): Expected minimum size: ${EXPECTED_SIZE}KB" >> /tmp/auto-expand.log

    if [ "$ROOT_SIZE" -lt "$EXPECTED_SIZE" ]; then
        echo "$(date): Partition expansion needed" >> /tmp/auto-expand.log
        echo "$(date): Current partition table:" >> /tmp/auto-expand.log
        fdisk -l /dev/mmcblk0 >> /tmp/auto-expand.log 2>&1

        # 확장 전 파티션2 섹터 수 저장
        ORIGINAL_SECTORS=$(fdisk -l /dev/mmcblk0 | grep mmcblk0p2 | awk '{print $4}')
        echo "$(date): Original partition 2 sectors: $ORIGINAL_SECTORS" >> /tmp/auto-expand.log

        # 파티션1의 끝 섹터 다음부터 시작하도록 동적 계산
        PART1_END=$(fdisk -l /dev/mmcblk0 | grep mmcblk0p1 | awk '{print $3}')
        PART2_START=$((PART1_END + 1))
        echo "$(date): Calculated partition 2 start sector: $PART2_START" >> /tmp/auto-expand.log

        # fdisk로 파티션 2 확장
        echo "$(date): Expanding partition 2" >> /tmp/auto-expand.log
        (
            echo d          # 파티션 삭제
            echo 2          # 파티션 2
            echo n          # 새 파티션 생성
            echo p          # primary
            echo 2          # 파티션 번호 2
            echo $PART2_START  # 동적으로 계산된 시작 섹터
            echo            # 기본 끝점 (디스크 끝)
            echo w          # 저장
        ) | fdisk /dev/mmcblk0 >> /tmp/auto-expand.log 2>&1

        # fdisk 완료 후 잠시 대기
        sleep 3

        # 확장 후 파티션2 섹터 수 확인
        NEW_SECTORS=$(fdisk -l /dev/mmcblk0 2>/dev/null | grep mmcblk0p2 | awk '{print $4}')
        echo "$(date): New partition 2 sectors: $NEW_SECTORS" >> /tmp/auto-expand.log

        # 섹터 수가 증가했는지 확인 (최소 2배 이상)
        if [ -n "$NEW_SECTORS" ] && [ -n "$ORIGINAL_SECTORS" ] && [ "$NEW_SECTORS" -gt $((ORIGINAL_SECTORS * 2)) ]; then
            echo "$(date): Partition expansion successful (sectors: $ORIGINAL_SECTORS -> $NEW_SECTORS)" >> /tmp/auto-expand.log
            touch /.filesystem_expanded
            sync
            echo "$(date): Rebooting to apply partition changes..." >> /tmp/auto-expand.log
            sleep 3
            /sbin/reboot
        else
            echo "$(date): Partition expansion failed or insufficient (sectors: $ORIGINAL_SECTORS -> $NEW_SECTORS)" >> /tmp/auto-expand.log
            echo "$(date): Will retry on next boot" >> /tmp/auto-expand.log
        fi
    else
        echo "$(date): Partition already large enough" >> /tmp/auto-expand.log
        touch /.filesystem_expanded
    fi
fi

# 두 번째 부팅 시 파일시스템 확장
if [ -f /.filesystem_expanded ] && [ ! -f /.filesystem_resized ]; then
    echo "$(date): Second boot - expanding filesystem" >> /tmp/auto-expand.log

    # 파일시스템 확장 전 상태
    echo "$(date): Filesystem size before expansion:" >> /tmp/auto-expand.log
    df -h / >> /tmp/auto-expand.log

    # 현재 파티션 섹터 수와 파일시스템 크기 확인
    CURRENT_SECTORS=$(fdisk -l /dev/mmcblk0 | grep mmcblk0p2 | awk '{print $4}')
    FILESYSTEM_SIZE_KB=$(df / | tail -1 | awk '{print $2}')
    PARTITION_SIZE_KB=$((CURRENT_SECTORS * 512 / 1024))  # 섹터를 KB로 변환

    echo "$(date): Partition size: ${PARTITION_SIZE_KB}KB (${CURRENT_SECTORS} sectors)" >> /tmp/auto-expand.log
    echo "$(date): Filesystem size: ${FILESYSTEM_SIZE_KB}KB" >> /tmp/auto-expand.log

    # 파티션이 파일시스템보다 50% 이상 크면 확장 실행
    EXPANSION_THRESHOLD=$((FILESYSTEM_SIZE_KB * 150 / 100))
    if [ "$PARTITION_SIZE_KB" -gt "$EXPANSION_THRESHOLD" ]; then
        echo "$(date): Partition is significantly larger than filesystem, expanding..." >> /tmp/auto-expand.log

        # ext4 파일시스템 확장
        resize2fs /dev/mmcblk0p2 >> /tmp/auto-expand.log 2>&1
        RESIZE_RESULT=$?

        echo "$(date): resize2fs result code: $RESIZE_RESULT" >> /tmp/auto-expand.log

        if [ $RESIZE_RESULT -eq 0 ]; then
            touch /.filesystem_resized
            echo "$(date): Filesystem expansion completed successfully" >> /tmp/auto-expand.log

            # 확장 후 상태
            echo "$(date): Filesystem size after expansion:" >> /tmp/auto-expand.log
            df -h / >> /tmp/auto-expand.log

            echo "$(date): Final partition table:" >> /tmp/auto-expand.log
            fdisk -l /dev/mmcblk0 >> /tmp/auto-expand.log 2>&1
        else
            echo "$(date): ERROR: resize2fs failed with code $RESIZE_RESULT" >> /tmp/auto-expand.log
        fi
    else
        echo "$(date): Partition size not significantly larger than filesystem, skipping expansion" >> /tmp/auto-expand.log
        echo "$(date): This might indicate the partition expansion didn't work properly" >> /tmp/auto-expand.log
        # 파티션 확장이 제대로 안된 경우를 위해 마커 파일 삭제
        rm -f /.filesystem_expanded
    fi
fi

# 확장 완료 상태 표시
if [ -f /.filesystem_expanded ] && [ -f /.filesystem_resized ]; then
    echo "$(date): Auto-expansion completed - system ready" >> /tmp/auto-expand.log

    # 최종 상태 확인
    FINAL_SIZE_KB=$(df / | tail -1 | awk '{print $2}')
    FINAL_SIZE_GB=$((FINAL_SIZE_KB / 1024 / 1024))
    DISK_SIZE_GB=$(($(fdisk -l /dev/mmcblk0 | grep "Disk /dev/mmcblk0" | awk '{print $5}') / 1024 / 1024 / 1024))
    echo "$(date): Final filesystem: ${FINAL_SIZE_GB}GB on ${DISK_SIZE_GB}GB disk" >> /tmp/auto-expand.log

    # 확장 성공률 계산
    USAGE_PERCENT=$((FINAL_SIZE_GB * 100 / DISK_SIZE_GB))
    echo "$(date): Disk utilization: ${USAGE_PERCENT}%" >> /tmp/auto-expand.log

    if [ "$USAGE_PERCENT" -gt 80 ]; then
        echo "$(date): SUCCESS: Disk space efficiently utilized" >> /tmp/auto-expand.log
    else
        echo "$(date): WARNING: Low disk utilization, expansion may be incomplete" >> /tmp/auto-expand.log
    fi
fi
